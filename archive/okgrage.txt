        // --- INDEXEDDB IMPLEMENTATION ---
        const DB_NAME = 'okGarageDB';
        const STORE_NAME = 'garageStore';

        const dbPromise = new Promise((resolve, reject) => {
            const request = indexedDB.open(DB_NAME, 1);
            request.onupgradeneeded = (event) => {
                const db = event.target.result;
                if (!db.objectStoreNames.contains(STORE_NAME)) {
                    db.createObjectStore(STORE_NAME);
                }
            };
            request.onsuccess = (event) => resolve(event.target.result);
            request.onerror = (event) => reject(event.target.error);
        });

        async function saveToDB(key, data) {
            const db = await dbPromise;
            return new Promise((resolve, reject) => {
                const tx = db.transaction(STORE_NAME, 'readwrite');
                const store = tx.objectStore(STORE_NAME);
                store.put(data, key);
                tx.oncomplete = () => resolve();
                tx.onerror = () => reject(tx.error);
            });
        }

        async function loadFromDB(key) {
            const db = await dbPromise;
            return new Promise((resolve, reject) => {
                const tx = db.transaction(STORE_NAME, 'readonly');
                const store = tx.objectStore(STORE_NAME);
                const request = store.get(key);
                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);
            });
        }

            showValuation(shouldOpen = false) {
                const list = document.getElementById('valuationList');
                if (!list) return;
                list.innerHTML = '';
                let totalBuy = 0;
                let totalSell = 0;

                // Filter Logic
                let displayCars = this.data.cars;
                const searchInput = document.getElementById('valSearchInput');
                if (searchInput && searchInput.value.trim() !== "") {
                    const term = searchInput.value.toLowerCase();
                    displayCars = displayCars.filter(c => c.name.toLowerCase().includes(term) || (c.real && c.real.toLowerCase().includes(term)));
                }

                // Sort by Buy Price Descending
                const sorted = [...displayCars].sort((a, b) => (b.buyPrice || 0) - (a.buyPrice || 0));

                sorted.forEach(c => {
                    const buy = parseInt(c.buyPrice) || 0;
                    const miles = c.miles || 0;
                    let sell = 0;

                    // Depreciation Logic
                    if (miles <= 1) {
                        sell = buy; // 100%
                    } else if (miles < 7) {
                        // Linear slide: 100% -> 87% over 6 miles
                        const progress = (miles - 1) / 6;
                        const factor = 1.0 - (progress * 0.13);
                        sell = Math.ceil(buy * factor);
                    } else {
                        sell = Math.ceil(buy * 0.87); // Standard 87%
                    }

                    totalBuy += buy;
                    totalSell += sell;

                    // Create Row

            refreshCarTasks() {
                if (!this.activeCar) return;

                // Clear any existing intervals
                if (this.activeCar.quests) {
                    this.activeCar.quests.forEach(q => {
                        if (this.timerIntervals[q.id]) clearInterval(this.timerIntervals[q.id]);
                    });
                }

                // WEIGHTED RANDOM SELECTION
                const pickWeighted = (excludeNames = []) => {
                    const pool = CAR_QUESTS.filter(q => !excludeNames.includes(q.name));
                    if (pool.length === 0) return CAR_QUESTS[0];

                    let totalWeight = 0;
                    pool.forEach(q => {
                        const w = (this.data.questWeights && this.data.questWeights[q.name]) ? this.data.questWeights[q.name] : 100;
                        totalWeight += w;
                    });

                    let random = Math.random() * totalWeight;
                    for (let i = 0; i < pool.length; i++) {
                        const qName = pool[i].name;
                        const weight = (this.data.questWeights && this.data.questWeights[qName]) ? this.data.questWeights[qName] : 100;
                        if (random < weight) return pool[i];
                        random -= weight;
                    }
                    return pool[0];
                };

                this.activeCar.quests = [];
                const pickedNames = [];

                for (let i = 0; i < 3; i++) {
                    const q = pickWeighted(pickedNames);
                    pickedNames.push(q.name);
                    this.activeCar.quests.push({
                        txt: q.name,
                        xp: q.xp,
                        done: false,
                        id: Date.now() + i,
                        type: q.type || 'simple',
                        duration: q.duration || 0,
                        countTarget: q.count || 0,
                        currentCount: 0,
                        timerRunning: false,
                        timeLeft: q.duration || 0,
                        label: q.label || "Start",
                        stepLabel: q.stepLabel || "Check"
                    });
                }
                // NEW: Save to LocalStorage Backup as well for redundancy
                try {
                    localStorage.setItem('okgarage_v1_backup', JSON.stringify(this.data));
                } catch (e) { console.warn("LS Backup failed (likely quota)", e); }

                this.save();
                this.renderCarTasks();
            }

            let timeBonus = 0;

            if (data.type === 'miles') {
                // Manual Complete for Miles
                isReady = true;
                progressPct = 100;
                progressText = "COMPLETE MISSION";
            } else if (data.type === 'drive' || data.type === 'timer') {

                const totalDur = Number(data.target) || 1800; // Total duration in seconds

                data.progress = elapsed; // Update progress

                const rem = Math.max(0, totalDur - elapsed);
                progressPct = Math.min(100, (elapsed / totalDur) * 100);

                // Time Bonus: 1 XP per second elapsed
                timeBonus = elapsed * 1;

                // Format Time: MM:SS
                const m = String(Math.floor(rem / 60)).padStart(2, '0');
                const s = String(rem % 60).padStart(2, '0');

                    progressPct = 100;

        completeActiveContract() {
            if (!this.data.activeContract) return;
            let reward = this.data.activeContract.xpReward || 500;

            // Time Bonus Logic (1 XP per Sec)
            // If Timer/Drive type, add bonus
            if (this.data.activeContract.type === 'drive' || this.data.activeContract.type === 'timer') {
                const totalDur = Number(this.data.activeContract.target) || 1800;
                const bonus = Math.floor(totalDur * 1);
                reward += bonus;
                ui.showToast(`Mission Complete! +${this.data.activeContract.xpReward} XP + ${bonus} Time Bonus`);
            }

            // NEW: Stalk/Crash Cumulative Bonus
            if (this.data.activeContract.title === 'Shadow Assignment' || this.data.activeContract.title === 'Zero Tolerance Run') {
                const bonus = this.data.activeContract.accumulatedXp || 0;
                reward += bonus;
                if (bonus > 0) ui.showToast(`Mission Complete! +${this.data.activeContract.xpReward} XP + ${bonus} Bonus`);
            }

            const oldXp = this.data.user.xp;
            const oldLvl = this.data.user.level;

            this.data.user.xp += reward;

            // Force Animation Overlay
            this.animateXpOverlay(oldXp, this.data.user.xp, oldLvl);

            this.data.activeContract = null;

            // HIDE WIDGET IMMEDIATELY
            const widget = document.getElementById('contractWidget');
            const details = document.getElementById('contractWidgetDetails');
            if (widget) widget.classList.add('hidden');
            if (details) details.classList.remove('active'); // Close details view
            widget.removeAttribute('data-animated'); // Reset animation flag

            this.save();

            // Refresh board if open to show "No Active Contract"
            const modal = document.getElementById('contractsModal');
            if (modal && modal.classList.contains('active')) this.renderContracts();
        }

        initWidgetDragLogic() {
            const widget = document.getElementById('contractWidget');
            if (!widget) return;

            // Reset positioning to avoid "stuck" states
            widget.style.transform = "none";
            widget.style.left = "20px";
            widget.style.bottom = "20px";
            widget.style.zIndex = "99999";
            widget.style.cursor = "grab";

            // Variables
            let isDragging = false;
            let startX, startY, initialLeft, initialBottom;
            let hasMoved = false;

            const onMouseDown = (e) => {
                if (e.button !== 0) return; // Left click only
                isDragging = true;
                hasMoved = false;

                // Get current computed position
                const style = window.getComputedStyle(widget);
                initialLeft = parseInt(style.left) || 0;
                initialBottom = parseInt(style.bottom) || 0;

                startX = e.clientX;
                startY = e.clientY;

                widget.style.transition = 'none';
                widget.style.cursor = "grabbing";

                // Prevent default to stop text selection
                e.preventDefault();
            };

            const onMouseMove = (e) => {
                if (!isDragging) return;

                const dx = e.clientX - startX;
                const dy = e.clientY - startY; // Dragging down increases Y

                // Invert DY for "bottom" positioning? No, if we use Top/Left it's easier.
                // Let's stick to Left/Bottom as defined in CSS usually, but let's switch to Left/Top for easier dragging math.
                // Actually, let's just use transform translate for performance? No, user wants simple.
                // Let's use Left/Top.

                // Force switch to Left/Top if not set
                if (widget.style.top === "") {
                    const rec = widget.getBoundingClientRect();
                    widget.style.bottom = "auto";
                    widget.style.top = rec.top + "px";
                    widget.style.left = rec.left + "px";
                    initialLeft = rec.left;
                    // initialTop... wait, we need new vars
                }

                // Okay simpler: simple standard drag impl using absolute position
                // We need to calculate new position
                // We can just use transforms.

                if (Math.abs(dx) > 3 || Math.abs(dy) > 3) hasMoved = true;

                widget.style.transform = `translate(${dx}px, ${dy}px)`;

                // Move Details Popup if active
                // Nested structure in HTML handles this now.
                // Details will move with the widget automatically.
            };

            const onMouseUp = (e) => {
                if (!isDragging) return;
                isDragging = false;
                widget.style.cursor = "grab";

                // Commit the transform to actual position or leave it?
                // If we leave it, next drag starts from 0 again.
                // Let's apply to Top/Left.
                const rect = widget.getBoundingClientRect();
                widget.style.transform = "none";
                widget.style.top = rect.top + "px";
                widget.style.left = rect.left + "px";
                widget.style.bottom = "auto";
                widget.style.right = "auto";

                if (!hasMoved) {
                    // It was a click!
                    const details = document.getElementById('contractWidgetDetails');
                    if (details) details.classList.toggle('active');
                }
            };

            // Touch support
            const onTouchStart = (e) => {
                const touch = e.touches[0];
                onMouseDown({ clientX: touch.clientX, clientY: touch.clientY, button: 0, preventDefault: () => e.preventDefault() });
            };
            const onTouchMove = (e) => {
                const touch = e.touches[0];
                onMouseMove({ clientX: touch.clientX, clientY: touch.clientY });
            };
            const onTouchEnd = (e) => {
                onMouseUp(e);
            };

            // Remove old listeners to prevent stacking (quick hack: replace element unique)
            // Better: just add if not present.
            if (widget.dataset.dragAttached) return;
            widget.dataset.dragAttached = "true";

            widget.addEventListener('mousedown', onMouseDown);
            window.addEventListener('mousemove', onMouseMove);
            window.addEventListener('mouseup', onMouseUp);

            widget.addEventListener('touchstart', onTouchStart, { passive: false });
            window.addEventListener('touchmove', onTouchMove, { passive: false });
            window.addEventListener('touchend', onTouchEnd);
        }

        repairJSON(txt) {
            txt = txt.replace(/"cars":\s*},\s*{/g, '"cars": [{');
            txt = txt.replace(/"quests":\s*}/g, '"quests": []}');
            txt = txt.replace(/"quests":\s*},/g, '"quests": []},');
            const open = (txt.match(/\{/g) || []).length;
            const close = (txt.match(/\}/g) || []).length;
            if (open > close) { txt += "}".repeat(open - close); }
            return txt;
        }

        renderGrid() {
            const grid = document.getElementById('carGrid');
            grid.innerHTML = "";

            // 1. Filter
            let matches = this.data.cars.filter(car => {
                const search = this.filters.search.toLowerCase();
                const nameMatch = car.name.toLowerCase().includes(search) || (car.real && car.real.toLowerCase().includes(search));
                const catMatch = this.filters.cat === "All" || car.cat === this.filters.cat;
                const passMatch = this.filters.pass === "All" || car.pass === this.filters.pass;
                const statusMatch = this.filters.status === "All" || (car.status && car.status.includes(this.filters.status));
                return nameMatch && catMatch && passMatch && statusMatch;
            });

            // 2. Separate Favorites
            let favs = matches.filter(c => c.favorite);
            let others = matches.filter(c => !c.favorite);

            // 3. Helper to sort a list based on current mode
            // 3. Helper to sort a list based on current mode
            const applySort = (list) => {
                const key = this.filters.sort;

                // Helper: Extract Year (19xx or 20xx) from Name or Real Name
                const getYear = (c) => {
                    const txt = c.real || c.name || "";
                    const match = txt.match(/\b(19|20)\d{2}\b/);
                    return match ? parseInt(match[0]) : null;
                };

                // Helper: Name Comparison (Case Insensitive)
                const compareName = (a, b) => (a.name || "").localeCompare(b.name || "");

                return list.sort((a, b) => {
                    const yearA = getYear(a);
                    const yearB = getYear(b);

                    if (key === "Newest") {
                        // Year Descending -> Name A-Z
                        if (yearA && yearB && yearA !== yearB) return yearB - yearA;
                        if (yearA && !yearB) return -1; // Year first
                        if (!yearA && yearB) return 1;
                        return compareName(a, b);
                    }
                    if (key === "Oldest") {
                        // Year Ascending -> Name A-Z
                        if (yearA && yearB && yearA !== yearB) return yearA - yearB;
                        if (yearA && !yearB) return -1; // Year first
                        if (!yearA && yearB) return 1;
                        return compareName(a, b);
                    }
                    if (key === "Highest XP") return (parseFloat(b.xp) || 0) - (parseFloat(a.xp) || 0);
                    if (key === "Lowest XP") return (parseFloat(a.xp) || 0) - (parseFloat(b.xp) || 0);
                    if (key === "Highest Price") return (parseFloat(b.sellPrice) || parseFloat(b.buyPrice) || 0) - (parseFloat(a.sellPrice) || parseFloat(a.buyPrice) || 0);
                    if (key === "Lowest Price") return (parseFloat(a.sellPrice) || parseFloat(a.buyPrice) || 0) - (parseFloat(b.sellPrice) || parseFloat(b.buyPrice) || 0);
                    if (key === "Highest Miles") return (parseFloat(b.miles) || 0) - (parseFloat(a.miles) || 0);
                    if (key === "Lowest Miles") return (parseFloat(a.miles) || 0) - (parseFloat(b.miles) || 0);

                    if (key === "A-Z + Year") {
                        // 1. No Year FIRST (A-Z)
                        if (!yearA && yearB) return -1;
                        if (yearA && !yearB) return 1;

                        // 2. Both No Year -> A-Z
                        if (!yearA && !yearB) return compareName(a, b);

                        // 3. Both Have Year -> Year Descending
                        if (yearA !== yearB) return yearB - yearA;

                        // 4. Same Year -> A-Z
                        return compareName(a, b);
                    }

                    // Default: A-Z
                    return compareName(a, b);
                });
            };

            // 4. Sort both groups independently
            // We clone arrays with [...list] to avoid mutating the original data structure during sort
            favs = applySort([...favs]);
            others = applySort([...others]);

            // 5. Merge (Favorites First)
            const finalDraw = favs.concat(others);

            finalDraw.forEach(car => {
                const el = document.createElement('div');

                // Safety Defaults
                car.miles = car.miles || 0;
                car.xp = car.xp || 0;
                car.incidents = car.incidents || 0;

                const favClass = car.favorite ? 'fav-active' : '';
                const lockClass = car.locked ? 'lock-active' : '';

                el.className = `car-card group ${favClass} ${lockClass}`;
                el.setAttribute('data-id', car.id);

                // Left Click: Open Detail
                el.onclick = () => {
                    if (!this.isImageEditMode && !car.locked) this.openDetail(car.id);
                    else if (car.locked) ui.showToast("Vehicle is Locked.");
                };

                // Right Click: Context Menu
                el.oncontextmenu = (e) => {
                    this.openContextMenu(e, car);
                };

                const showReal = car.real && car.real.trim() !== "" && car.real !== car.name;
                const posStyle = car.imgPos ? `object-position: ${car.imgPos}` : '';

                let tagsHtml = '';
                if (car.status) {
                    const tags = car.status.split(',').map(t => t.trim()).filter(t => t);
                    tags.forEach(tag => {
                        const cls = tag.toLowerCase().includes('limited') ? 'limited' :
                            tag.toLowerCase().includes('offsale') ? 'offsale' :
                                tag.toLowerCase().includes('collectible') ? 'collectible' : '';
                        tagsHtml += `<span class="tag-pill ${cls}">${tag}</span>`;
                    });
                }

                const centerLock = car.locked ? `<div class="absolute inset-0 z-30 flex items-center justify-center pointer-events-none"><i class="fa-solid fa-lock text-6xl text-white/20"></i></div>` : '';
                const cornerFav = car.favorite ? `<div class="absolute top-2 right-2 z-30"><i class="fa-solid fa-star text-yellow-400 drop-shadow-md text-xl"></i></div>` : '';

                el.innerHTML = `
                        ${centerLock}
                        ${cornerFav}

                        ${car.img ? `<img src="${car.img}" class="bg-img" style="${posStyle}">` : (car.locked ? `<div class="bg-img bg-white/5 flex items-center justify-center h-full w-full"></div>` : `<div class="bg-img bg-white/5 flex items-center justify-center h-full w-full"><i class="fa-solid fa-car text-4xl opacity-20"></i></div>`)}
                        <div class="card-gradient"></div>
                        
                        <div class="card-content">
                            <div class="text-group">
                                <div class="card-header-row"> <!-- Layout: Name Left, Tags Right -->
                                    <div class="card-name font-black italic tracking-tighter uppercase text-white shadow-black drop-shadow-md">

                            </div>

        // Updated handleImageUpload to use full quality/larger images now that we have IDB
        // Still compressing slightly (2048px width) to be reasonable, but much better than 800px
        handleImageUpload(input) {
            const file = input.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = (e) => {
                const img = new Image();
                img.onload = () => {
                    const canvas = document.getElementById('compressCanvas');
                    const ctx = canvas.getContext('2d');

                    // Resize logic: scale down to max width 2048px (high quality)
                    const scale = Math.min(2048 / img.width, 1);
                    canvas.width = img.width * scale;
                    canvas.height = img.height * scale;

                    ctx.drawImage(img, 0, 0, canvas.width, canvas.height);

                    // Get compressed base64 (JPEG quality 0.9 - High)
                    const base64 = canvas.toDataURL('image/jpeg', 0.9);

                    document.getElementById('inpImg').value = base64;
                    this.updatePreview();
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);
            input.value = ''; // Reset input so same file can be selected again
        }

        initCustomDropdowns() {
            const dropdowns = document.querySelectorAll('.custom-dropdown');
            dropdowns.forEach(dd => {
                const trigger = dd.querySelector('.dropdown-trigger');
                const options = dd.querySelectorAll('.dropdown-option');

                // Toggle dropdown
                trigger.addEventListener('click', (e) => {
                    e.stopPropagation();
                    // Close others
                    dropdowns.forEach(other => {
                        if (other !== dd) other.classList.remove('open');
                    });
                    dd.classList.toggle('open');
                });

                // Select option
                options.forEach(opt => {
                    opt.addEventListener('click', (e) => {
                        e.stopPropagation();
                        const val = opt.getAttribute('data-value');
                        const html = opt.innerHTML; // Get HTML to preserve icons

                        // Visual update
                        trigger.innerHTML = html;
                        options.forEach(o => o.classList.remove('selected'));
                        opt.classList.add('selected');
                        dd.classList.remove('open');

                        // Update hidden native select
                        let nativeSelectId = "";
                        if (dd.id.startsWith('ddFilter')) {
                            nativeSelectId = dd.id.replace('dd', '');
                            nativeSelectId = nativeSelectId.charAt(0).toLowerCase() + nativeSelectId.slice(1);
                        } else {
                            nativeSelectId = dd.id.replace('dd', 'inp');
                        }

                        const nativeSelect = document.getElementById(nativeSelectId);
                        if (nativeSelect) {
                            nativeSelect.value = val;
                            // Trigger change event manually for filters to update instantly
                            nativeSelect.dispatchEvent(new Event('change'));
                        }
                    });
                });
            });

            // Close on outside click
            window.addEventListener('click', () => {
                dropdowns.forEach(dd => dd.classList.remove('open'));
            });
        }

        toggleQuest(carId, questIdx, btn) {
            // Prevent event bubbling if needed
            // event.stopPropagation();

            const car = this.data.cars.find(c => c.id === carId);
            if (!car || !car.quests[questIdx]) return;

            // Toggle
            car.quests[questIdx].done = !car.quests[questIdx].done;

            // XP Logic
            if (car.quests[questIdx].done) {
                const xp = car.quests[questIdx].xp || 0;
                car.xp += xp;
                this.gainGlobalXp(xp);
                ui.showToast(`Quest Complete! +${xp} XP`);
                btn.innerText = "DONE";
                btn.style.opacity = "0.5";
            } else {
                const xp = car.quests[questIdx].xp || 0;
                car.xp -= xp; // Remove XP if unchecked
                if (car.xp < 0) car.xp = 0;
                btn.innerText = "DO";
                btn.style.opacity = "1";
            }

            this.save();
            this.updateCarXpUI();
        }

        saveCar() {
            const id = document.getElementById('inpId').value;
            const name = document.getElementById('inpName').value;
            const real = document.getElementById('inpReal').value;

            // Dropdowns
            const cat = document.getElementById('inpCat').value;
            const pass = document.getElementById('inpPass').value;

            // Specs
            const img = document.getElementById('inpImg').value;
            const trim = document.getElementById('inpTrim').value;
            const color = document.getElementById('inpColor').value;
            const plate = document.getElementById('inpPlate').value;
            const hp = document.getElementById('inpHp').value;
            const speed = document.getElementById('inpSpeed').value;
            const accel = document.getElementById('inp060').value;
            const milesVal = parseFloat(document.getElementById('inpMiles').value) || 0;

            // NEW FIELDS
            const weight = document.getElementById('inpWeight').value;
            const seats = document.getElementById('inpSeats').value;
            const fuel = document.getElementById('inpFuel').value;
            const country = document.getElementById('inpCountry').value;
            const engine = document.getElementById('inpEngine').value;
            const trans = document.getElementById('inpTrans').value; // from hidden select

            const safety = document.getElementById('inpSafety').value;
            const tow = document.getElementById('inpTow').value === "true";
            const buyPrice = parseFloat(document.getElementById('inpBuy').value) || 0;
            // Sell Price Removed - Calculated dynamically now
            const status = document.getElementById('inpStatus').value;
            const imgPos = document.getElementById('inpPreview').style.objectPosition || '50% 50%';

            if (id) {
                const idx = this.data.cars.findIndex(c => c.id === id);
                if (idx > -1) {
                    const car = this.data.cars[idx];
                    const oldMiles = car.miles || 0;
                    const diff = milesVal - oldMiles;

                    car.name = name; car.real = real; car.cat = cat; car.pass = pass;
                    car.img = img; car.trim = trim; car.color = color; car.plate = plate;
                    car.hp = hp;
                    car.speed = speed;
                    car.accel = accel;
                    car.safety = safety;
                    car.tow = tow;
                    car.buyPrice = buyPrice;
                    // car.sellPrice = sellPrice; // Removed
                    car.status = status;
                    car.imgPos = imgPos;

                    // SAVE NEW FIELDS
                    car.weight = weight;
                    car.seats = seats;
                    car.fuel = fuel;
                    car.country = country;
                    car.engine = engine;
                    car.trans = trans;
                    if (diff !== 0) {
                        car.miles = milesVal;
                        const xpDiff = Math.floor(diff * 5);
                        car.xp = (car.xp || 0) + xpDiff;
                        if (car.xp < 0) car.xp = 0;
                        this.gainGlobalXp(xpDiff);
                    } else {
                        car.miles = milesVal;
                    }
                }
            } else {
                const newCar = {
                    id: crypto.randomUUID(), name, real, cat, pass,
                    img, trim, color, plate, hp, speed, accel,
                    miles: milesVal, incidents: 0, xp: 0,
                    safety, tow, buyPrice, status, imgPos,
                    weight, seats, fuel, country, engine, trans, // NEW FIELDS
                    quests: [],
                    gallery: []
                };
                this.data.cars.push(newCar);
                ui.showToast('Vehicle Added');
            }
            this.save(true);
            this.renderGrid();
            ui.toggleModal('addModal');
            if (this.activeCar && this.activeCar.id === id) { this.openDetail(id); }
        }

        initCustomDropdowns() {
            const dropdowns = document.querySelectorAll('.custom-dropdown');
            dropdowns.forEach(dd => {
                const trigger = dd.querySelector('.dropdown-trigger');
                const options = dd.querySelectorAll('.dropdown-option');

                trigger.addEventListener('click', (e) => {
                    e.stopPropagation();
                    dropdowns.forEach(other => { if (other !== dd) other.classList.remove('open'); });
                    dd.classList.toggle('open');
                });

                options.forEach(opt => {
                    opt.addEventListener('click', (e) => {
                        e.stopPropagation();
                        const val = opt.getAttribute('data-value');
                        const html = opt.innerHTML;
                        trigger.innerHTML = html;
                        options.forEach(o => o.classList.remove('selected'));
                        opt.classList.add('selected');
                        dd.classList.remove('open');

                        // Sync with hidden native select
                        let selectId = dd.id.replace('dd', 'inp');
                        // Handle filter modal IDs differently
                        if (dd.id.includes('Filter')) selectId = dd.id.replace('dd', '').replace('Cat', 'Cat').replace('Pass', 'Pass').replace('Status', 'Status');
                        if (dd.id === 'ddFilterCat') selectId = 'filterCat';

                        const nativeSelect = document.getElementById(selectId);
                        if (nativeSelect) {
                            nativeSelect.value = val;
                            nativeSelect.dispatchEvent(new Event('change'));
                        }
                    });
                });
            });
            window.addEventListener('click', () => { dropdowns.forEach(dd => dd.classList.remove('open')); });
        }

        initDragLogic() {
            window.addEventListener('mousedown', (e) => {
                // 1. MANAGE VEHICLE MODAL (Always Active)
                const box = e.target.closest('#imgEditBox');
                if (box) {
                    e.preventDefault();
                    const img = box.querySelector('img');
                    if (!img) return;
                    this.drag.active = true;
                    this.drag.startX = e.clientX;
                    this.drag.startY = e.clientY;
                    this.drag.imgEl = img;
                    this.drag.targetId = 'PREVIEW';
                    // Parse current position
                    const style = img.style.objectPosition || '50% 50%';
                    const parts = style.split(' ');
                    this.drag.initialPos = { x: parseFloat(parts[0]) || 50, y: parseFloat(parts[1]) || 50 };
                    return;
                }

                // 2. MAIN GRID (Only if Edit Mode is ON)
                if (this.isImageEditMode) {
                    const card = e.target.closest('.car-card');
                    if (!card) return;
                    const img = card.querySelector('.bg-img');
                    if (!img) return;

                    e.preventDefault();
                    this.drag.active = true;
                    this.drag.startX = e.clientX;
                    this.drag.startY = e.clientY;
                    this.drag.targetId = card.getAttribute('data-id');
                    this.drag.imgEl = img;

                    const style = img.style.objectPosition || '50% 50%';
                    const parts = style.split(' ');
                    this.drag.initialPos = { x: parseFloat(parts[0]) || 50, y: parseFloat(parts[1]) || 50 };
                }
            });

            window.addEventListener('mousemove', (e) => {
                if (!this.drag.active || !this.drag.imgEl) return;
                e.preventDefault();

                const dx = e.clientX - this.drag.startX;
                const dy = e.clientY - this.drag.startY;

                // Sensitivity: Lower is slower/smoother
                const sensitivity = 0.25;

                let newX = this.drag.initialPos.x - (dx * sensitivity);
                let newY = this.drag.initialPos.y - (dy * sensitivity);

                // Clamp between 0% and 100%
                newX = Math.max(0, Math.min(100, newX));
                newY = Math.max(0, Math.min(100, newY));

                this.drag.imgEl.style.objectPosition = `${newX}% ${newY}%`;

                // Save immediately if not preview
                if (this.drag.targetId !== 'PREVIEW') {
                    const car = this.data.cars.find(c => c.id === this.drag.targetId);
                    if (car) car.imgPos = `${newX}% ${newY}%`;
                }
            });

            window.addEventListener('mouseup', () => {
                this.drag.active = false;
                this.drag.imgEl = null;
            });
        }

                const c = this.data.activeContract;
                const carName = typeof c.mutation.box2 === 'object' ? c.mutation.box2.name : c.mutation.box3;

                // Progress Logic
                let progressPct = 0;
                let progressText = "";
                let isReady = false;
                const target = Number(c.target) || 1;
                const current = c.progress || 0;

                if (c.type === 'timer' || c.type === 'drive') {
                    const duration = Number(c.target) || 1800;
                    let elapsed = current;
                    if (c.startTime) {
                        elapsed = Math.floor((Date.now() - c.startTime) / 1000);
                    }
                    progressPct = (elapsed / duration) * 100;
                    const rem = Math.max(0, duration - elapsed);
                    const m = String(Math.floor(rem / 60)).padStart(2, '0');
                    const s = String(rem % 60).padStart(2, '0');
                    progressText = `${m}:${s}`;
                    if (rem <= 0) { isReady = true; progressText = "COMPLETE MISSION"; progressPct = 100; }
                } else if (c.type === 'miles') {
                    // For Miles: Manual Complete Button (No tracking)
                    isReady = true;
                    progressText = "COMPLETE MISSION";
                    progressPct = 100;
                } else {
                    progressPct = (current / target) * 100;
                    progressText = `${current} / ${target}`;
                    if (current >= target) { isReady = true; progressText = "COMPLETE MISSION"; progressPct = 100; }

                                        <span class="z-10 relative text-right">${progressText}</span>
                                        <div class="tp-fill" style="width: ${progressPct}%"></div>
                                     </div>`
                                }
                            </div>
                        </div>
                    `;
                    
                    // Show message if hidden
                    document.getElementById('noActiveContractMsg').style.display = 'none';
                }
            } else {
                if (noContractMsg) noContractMsg.style.display = 'block';
            }
        }

                incrementContractCounter() {
                    const c = this.data.activeContract;
                    if (c && (c.type === 'counter' || c.type === 'miles' || c.type === 'simple')) {
                        // Ensure progress is initialized
                        c.progress = c.progress || 0;
                        const target = Number(c.target) || 1;

                        if (c.progress < target) {
                            c.progress++;
                            this.save();

                            // SMOOTH UPDATE: Don't re-render entire board, just update the pill
                            this.updateContractProgressUI();

                            // If the widget is open, update it
                            this.renderActiveContractWidget();

                            if (c.progress >= target) {
                                ui.showToast("Mission Objective Complete!");
                                // Re-render to show "COMPLETE MISSION" button state
                                this.renderContracts();
                            } else {
                                // Feedback for increment
                                // ui.showToast(`Progress: +1(${ c.progress } / ${ target })`, "fa-arrow-up");
                            }
                        }
                    }
                }

                handleStalkClick() {
                    const c = this.data.activeContract;
                    if (!c || c.title !== 'Shadow Assignment') return;

                    c.progress = c.progress || 0;
                    if (c.progress < 2) {
                        c.progress++;
                        if (c.progress === 1) ui.showToast("Tracking active...");
                        if (c.progress >= 2) ui.showToast("Target Stopped/Crashed!");

                        this.save();
                        this.renderContracts();
                    }
                }



                // NEW: Helper for smooth progress animation without re-rendering
                updateContractProgressUI() {
                    const c = this.data.activeContract;
                    if (!c) return;

                    // Find elements in Board
                    const boardPillFill = document.querySelector('#activeContractBoardDisplay .progress-pill-fill');
                    const boardPillText = document.querySelector('#activeContractBoardDisplay .progress-pill-text');

                    if (boardPillFill && boardPillText) {
                        let progressPct = 0;
                        let progressText = "";
                        const target = Number(c.target) || 1;
                        const current = c.progress || 0;

                        if (c.type === 'miles') {
                            // Inherit logic from renderContracts
                            const left = (target - current).toFixed(1);
                            progressText = `${ left } MILES LEFT`;
                            progressPct = (current / target) * 100;
                        } else {
                            progressPct = (current / target) * 100;
                            progressText = `${ current } / ${target}`;
                        }

                        // Update Style (CSS Transition handles smoothing)
                        boardPillFill.style.width = `${progressPct}%`;
                        boardPillText.innerHTML = progressText;
                    }
                }
            }
